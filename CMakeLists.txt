cmake_minimum_required(VERSION 3.16)
project(bale_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(BaleLoaderHydraulics SHARED src/LoaderHydraulics.cc)
target_link_libraries(BaleLoaderHydraulics PRIVATE
  gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
target_compile_options(BaleLoaderHydraulics PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(hydraulic_model_test tests/hydraulic_model_test.cc)
  target_include_directories(hydraulic_model_test PRIVATE src)
  add_test(NAME hydraulic_model COMMAND hydraulic_model_test)
endif()
