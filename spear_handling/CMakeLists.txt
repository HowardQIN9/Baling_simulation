cmake_minimum_required(VERSION 3.16)
project(tractor_spear_handling LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
find_package(gz-physics7 REQUIRED)
add_library(BaleLoaderHydraulics SHARED ../src/LoaderHydraulics.cc)
target_link_libraries(BaleLoaderHydraulics PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
add_library(SpearContact SHARED src/SpearContact.cc)
target_link_libraries(SpearContact PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13 gz-physics7::gz-physics7)
target_compile_options(SpearContact PRIVATE -Wall -Wextra -Wpedantic)
